販売状況リセットプログラム（GASに設定するやつ）
function setStatusByStoreWithVariableItems() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const storeCount = sheet.getRange("A4").getValue(); // 店舗数
  const startRow = 8;

  for (let i = 0; i < storeCount; i++) {
    const itemCountCol = 2 + i * 4;
    const itemCount = sheet.getRange(6, itemCountCol).getValue();
    const statusCol = itemCountCol + 1;
    const range = sheet.getRange(startRow, statusCol, itemCount);
    range.setValue("販売中");
  }
}
